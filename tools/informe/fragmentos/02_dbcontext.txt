using CatalogoVideojuegos.Models;   // Acceso a las clases del modelo.
using Microsoft.EntityFrameworkCore; // DbContext, DbSet, ModelBuilder.

namespace CatalogoVideojuegos.Data;

/// <summary>
/// Contexto de Entity Framework Core: representa la sesión con la base
/// de datos PostgreSQL y expone un DbSet por cada entidad.
/// </summary>
public class ApplicationDbContext : DbContext
{
    // El constructor recibe las opciones (cadena de conexión y proveedor)
    // que Program.cs configuró. EF Core lo instancia por inyección de
    // dependencias en cada controlador.
    public ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
        : base(options)
    {
    }

    // -----------------------------------------------------------
    // DbSet: una "puerta de entrada" a cada tabla.
    // EF Core traduce _context.Videojuegos a SELECT ... FROM "Videojuegos".
    // El nombre de la propiedad (plural) se usa como nombre de tabla.
    // -----------------------------------------------------------
    public DbSet<Videojuego> Videojuegos { get; set; }

    public DbSet<Plataforma> Plataformas { get; set; }

    public DbSet<Desarrolladora> Desarrolladoras { get; set; }

    public DbSet<Genero> Generos { get; set; }

    /// <summary>
    /// Configura las relaciones y reglas que no se pueden expresar con
    /// atributos. Se ejecuta una sola vez al construir el modelo.
    /// </summary>
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // -----------------------------------------------------------
        // RELACIÓN 1:N — Genero (principal) → Videojuego (dependiente)
        // HasOne / WithMany declaran la cardinalidad.
        // HasForeignKey indica qué propiedad es la llave foránea.
        // OnDelete(Restrict) genera ON DELETE RESTRICT: PostgreSQL
        // impedirá borrar un género que todavía tenga videojuegos.
        // -----------------------------------------------------------
        modelBuilder.Entity<Videojuego>()
            .HasOne(v => v.Genero)          // Un videojuego tiene un género.
            .WithMany(g => g.Videojuegos)   // Un género tiene muchos videojuegos.
            .HasForeignKey(v => v.GeneroId)// La FK es Videojuego.GeneroId.
            .OnDelete(DeleteBehavior.Restrict);

        // -----------------------------------------------------------
        // RELACIÓN 1:N — Desarrolladora (principal) → Videojuego
        // Misma configuración: protege el catálogo para que no se borren
        // videojuegos en cascada al eliminar un estudio.
        // -----------------------------------------------------------
        modelBuilder.Entity<Videojuego>()
            .HasOne(v => v.Desarrolladora)
            .WithMany(d => d.Videojuegos)
            .HasForeignKey(v => v.DesarrolladoraId)
            .OnDelete(DeleteBehavior.Restrict);

        // La relación N:N Videojuego ↔ Plataforma no se configura aquí:
        // EF Core la detecta por las colecciones ICollection de ambas
        // clases y genera la tabla intermedia en la migración.
    }
}
