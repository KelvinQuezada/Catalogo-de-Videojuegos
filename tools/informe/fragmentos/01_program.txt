using CatalogoVideojuegos.Data;          // Namespace del ApplicationDbContext.
using Microsoft.EntityFrameworkCore;      // Métodos de extensión: UseNpgsql.
using System.Globalization;               // CultureInfo para el formato numérico.

// ---------------------------------------------------------------
// 1. CULTURA NUMÉRICA
// Los formularios HTML usan type="number", que siempre trabaja con
// punto decimal. Si la cultura del servidor fuera es-EC, un precio
// como 19.99 se mostraría "19,99" y el navegador lo rechazaría.
// Se fija la cultura de datos a InvariantCulture y solo la interfaz
// (fechas, textos) queda en español de Ecuador.
// ---------------------------------------------------------------
CultureInfo.DefaultThreadCurrentCulture = CultureInfo.InvariantCulture;
CultureInfo.DefaultThreadCurrentUICulture = new CultureInfo("es-EC");

// Crea el constructor de la aplicación con la configuración de
// appsettings.json, variables de entorno y argumentos.
var builder = WebApplication.CreateBuilder(args);

// ---------------------------------------------------------------
// 2. SERVICIOS MVC
// Registra en el contenedor de dependencias el soporte para
// controladores y vistas Razor (.cshtml).
// ---------------------------------------------------------------
builder.Services.AddControllersWithViews();

// ---------------------------------------------------------------
// 3. REGISTRO DEL DbContext
// AddDbContext registra ApplicationDbContext con ámbito (scoped):
// se crea una instancia por cada petición HTTP y se destruye al
// terminar. UseNpgsql indica el proveedor de PostgreSQL y recibe la
// cadena de conexión leída de la sección ConnectionStrings.
// ---------------------------------------------------------------
builder.Services.AddDbContext<ApplicationDbContext>(options =>
    options.UseNpgsql(builder.Configuration.GetConnectionString("CatalogoVideojuegosDb")));

var app = builder.Build();

// En producción se muestra una página de error genérica y HSTS.
if (!app.Environment.IsDevelopment())
{
    app.UseExceptionHandler("/Home/Error");   // Ruta de error amigable.
    app.UseHsts();                            // Fuerza HTTPS en el navegador.
}

app.UseHttpsRedirection();   // Redirige HTTP → HTTPS cuando existe puerto seguro.
app.UseRouting();            // Habilita el sistema de rutas.

app.UseAuthorization();      // Sin Identity ni roles: no restringe nada,
                             // pero mantiene el orden correcto del pipeline.

app.MapStaticAssets();       // Sirve wwwroot (CSS, JS, imágenes) con caché.

// ---------------------------------------------------------------
// 4. RUTA POR DEFECTO
// /Videojuegos/Details/5 se traduce a:
//   controller = Videojuegos, action = Details, id = 5
// Si la URL está vacía se usa Home/Index.
// ---------------------------------------------------------------
app.MapControllerRoute(
    name: "default",
    pattern: "{controller=Home}/{action=Index}/{id?}")
    .WithStaticAssets();

app.Run();   // Inicia el servidor Kestrel y queda escuchando peticiones.
