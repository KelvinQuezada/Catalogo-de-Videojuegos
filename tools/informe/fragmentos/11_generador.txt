// ---------------------------------------------------------------------------
// FUNCIÓN CLAVE: decide en qué plataformas salió cada juego.
// Regla de negocio: un juego de Nintendo solo sale en consolas Nintendo;
// el resto depende del año de lanzamiento (generación de consolas) y del
// tamaño del estudio (los indie llegan a Steam Deck).
// ---------------------------------------------------------------------------
function plataformasDe(v) {
  const [titulo, genero, dev, , fecha] = v;      // Desestructuración de la tupla.
  const year = Number(fecha.slice(0, 4));        // Año de lanzamiento.
  const set = new Set();                         // Set evita ids repetidos.

  if (dev === 1) {                               // Nintendo EPD.
    set.add(6);                                  // Nintendo Switch.
    if (year >= 2025) set.add(7);                // Nintendo Switch 2.
    return [...set];
  }

  if (year <= 2012) {                            // Juegos antiguos.
    set.add(1); set.add(11); set.add(12);        // PC, Xbox 360, PS3.
  } else if (year <= 2015) {                     // Transición de generación.
    set.add(1); set.add(3); set.add(5);          // PC, PS4, Xbox One.
    if (year <= 2014) set.add(11);               // Todavía había Xbox 360.
  } else if (year <= 2019) {                     // Generación PS4/Xbox One.
    set.add(1); set.add(3); set.add(5);
    if (dev === 8 || dev >= 19) set.add(6);      // Valve e indie → Switch.
  } else if (year <= 2021) {                     // Año de lanzamiento de PS5.
    set.add(1); set.add(2); set.add(3); set.add(4); set.add(5);
  } else {                                       // 2022 en adelante.
    set.add(1); set.add(2); set.add(4);          // PC, PS5, Xbox Series.
    if (dev >= 18) set.add(6);                   // Estudios medianos → Switch.
    if (dev >= 19) set.add(8);                   // Indie → Steam Deck.
  }

  return [...set].sort((a, b) => a - b);         // Orden estable de ids.
}
